/** Throwaway: touch, gestures and back findings for the Phase 0 Perfect run. Never committed. */
@RunWith(RobolectricTestRunner::class)
@GraphicsMode(GraphicsMode.Mode.NATIVE)
@Config(application = Application::class, sdk = [36])
class ZzTouchTest {
    @get:Rule
    val rule = createAndroidComposeRule<ComponentActivity>()

    private val now = Instant.parse("2026-10-27T08:00:00Z").toEpochMilli()

    private val lib = Library(
        Json.parseToJsonElement(
            """
            {"theme":"midnight","bgGrad":30,
             "games":[
               {"id":"a","title":"Starfall Lantern","state":"backlog","date":"2026-09-30","ord":1,"colour":"#3A6FD8","genres":["Action","RPG"],"hours":22,"added":"2026-10-20"},
               {"id":"c","title":"Copper Meadow","state":"playing","date":"2026-05-11","ord":3,"colour":"#2E9E6A","genres":["Action"],"hours":12,"added":"2026-06-14"}
             ],"duels":[],"played":[]}
            """.trimIndent(),
        ).jsonObject,
    )
    private val ready = Games.State.Ready(lib, copyAt = now - 3_600_000L)

    @Suppress("UNCHECKED_CAST")
    private val games = Games::class.java.getDeclaredField("_state").apply { isAccessible = true }.get(null) as MutableStateFlow<Games.State>

    @Before
    fun setUp() {
        Store.init(ApplicationProvider.getApplicationContext())
        TimeZone.setDefault(TimeZone.getTimeZone("UTC"))
    }

    @After
    fun tearDown() {
        games.value = Games.State.Starting
    }

    /** "swipe-down closes any sheet" (Decisions, Oct 26): a drag down from Settings' grab handle. */
    @Test fun settingsSwipeDown() {
        RuntimeEnvironment.setQualifiers("w412dp-h915dp-xxhdpi")
        var nav by mutableStateOf(Nav("home", listOf("settings")))
        rule.setContent { Frame(Look.of(lib), Prefs.of(lib), now = { now }) { Shell(nav, ready, go = { nav = it }) } }
        val t = rule.onNodeWithText("Settings").fetchSemanticsNode().boundsInRoot
        println("ZZ sheet title at $t")
        rule.onRoot().performTouchInput { swipe(Offset(t.center.x, t.top + 30f), Offset(t.center.x, t.top + 900f), 250) }
        rule.waitForIdle()
        println("ZZ after swipe-down on the handle: stack=${nav.stack}")
        org.junit.Assert.assertTrue("a swipe down left Settings open: ${nav.stack}", nav.stack.isEmpty())
    }

    /** Back on the restore screen after Settings › "Use a different restore code". */
    @Test fun restoreBackAfterDifferentCode() {
        RuntimeEnvironment.setQualifiers("w412dp-h915dp-xxhdpi")
        games.value = ready
        rule.setContent { PlayNextRoot(null) {} }
        rule.onNodeWithContentDescription("Settings").performClick()
        rule.onNodeWithText("Use a different restore code").performClick()
        rule.waitUntil(5_000) { games.value is Games.State.NoCode }
        rule.waitForIdle()
        println("ZZ restore screen up: ${rule.onAllNodesWithText("Bring your games").fetchSemanticsNodes().size}")
        val d = rule.activity.onBackPressedDispatcher
        println("ZZ before back 1: enabled callbacks=${d.hasEnabledCallbacks()}")
        rule.runOnIdle { d.onBackPressed() }
        rule.waitForIdle()
        val still = rule.onAllNodesWithText("Bring your games").fetchSemanticsNodes().size
        println("ZZ after back 1: restore screen nodes=$still, enabled callbacks=${d.hasEnabledCallbacks()} (false = the next back leaves the app)")
        // what is under it: Not now shows it
        rule.onNodeWithText("Not now").performClick()
        rule.waitForIdle()
        val settingsUp = rule.onAllNodesWithText("Done").fetchSemanticsNodes().size
        val restoreUp = rule.onAllNodesWithText("Bring your games").fetchSemanticsNodes().size
        println("ZZ after Not now: Settings sheet nodes=$settingsUp, restore nodes=$restoreUp")
        org.junit.Assert.assertTrue("back on the restore screen changed nothing visible and the next back leaves the app", still == 0 && d.hasEnabledCallbacks())
    }

    /** "Not now" after Settings › "Use a different restore code": one tap. */
    @Test fun restoreNotNowAfterDifferentCode() {
        RuntimeEnvironment.setQualifiers("w412dp-h915dp-xxhdpi")
        games.value = ready
        rule.setContent { PlayNextRoot(null) {} }
        rule.onNodeWithContentDescription("Settings").performClick()
        rule.onNodeWithText("Use a different restore code").performClick()
        rule.waitUntil(5_000) { games.value is Games.State.NoCode }
        rule.waitForIdle()
        rule.onNodeWithText("Not now").performClick()
        rule.waitForIdle()

    /** Unfolded: two quick taps on the rail's Settings. */
    @Test fun railSettingsDoubleTap() {
        RuntimeEnvironment.setQualifiers("w859dp-h1000dp-xxhdpi")
        var nav by mutableStateOf(Nav("home"))
        rule.setContent { Frame(Look.of(lib), Prefs.of(lib), now = { now }) { Shell(nav, ready, go = { nav = it }) } }
        val g = rule.onNodeWithContentDescription("Settings").fetchSemanticsNode().boundsInRoot
        rule.onRoot().performTouchInput { click(g.center) }
        rule.mainClock.advanceTimeBy(120)
        println("ZZ after tap 1: stack=${nav.stack}")
        rule.onRoot().performTouchInput { click(g.center) }
        rule.waitForIdle()
        println("ZZ after tap 2 (120ms later): stack=${nav.stack}")
        org.junit.Assert.assertTrue("a double tap on the rail's Settings opened and closed it: ${nav.stack}", nav.stack == listOf("settings"))
    }

    /** Folded: two quick taps on Settings' Done, with the page scrolled to the top. */
    @Test fun doneDoubleTap() {
        RuntimeEnvironment.setQualifiers("w412dp-h915dp-xxhdpi")
        var nav by mutableStateOf(Nav("home", listOf("settings")))
        rule.setContent { Frame(Look.of(lib), Prefs.of(lib), now = { now }) { Shell(nav, ready, go = { nav = it }) } }
        val done = rule.onNodeWithText("Done").fetchSemanticsNode().boundsInRoot
        val gear = rule.onNodeWithContentDescription("Settings").fetchSemanticsNode().boundsInRoot
        val hit = done.intersect(gear)
        println("ZZ Done=$done gear=$gear overlap=$hit")
        val at = if (hit.width > 0 && hit.height > 0) hit.center else done.center
        rule.onRoot().performTouchInput { click(at) }
        rule.mainClock.advanceTimeBy(120)
        println("ZZ after tap 1: stack=${nav.stack}")
        rule.onRoot().performTouchInput { click(at) }
        rule.waitForIdle()
        println("ZZ after tap 2 (120ms later): stack=${nav.stack}")
        org.junit.Assert.assertTrue("a double tap on Done closed Settings and opened it again: ${nav.stack}", nav.stack.isEmpty())
    }

    /** The pill's lit segment: a second pick while it still slides. Where is the segment's left edge each frame? */
    @Test fun pillSecondPickMidSlide() {
        RuntimeEnvironment.setQualifiers("w412dp-h915dp-xxhdpi")
        var nav by mutableStateOf(Nav("games"))
        val look = Look.of(lib)
        rule.setContent { Frame(look, Prefs.of(lib), now = { now }) { Shell(nav, ready, go = { nav = it }) } }
        rule.waitForIdle()
        val volt = look.volt
        fun btn(label: String) = rule.onAllNodesWithContentDescription(label).fetchSemanticsNodes().minBy { it.size.height }.boundsInRoot
        val labels = Prefs.of(lib).views("games").map { it.second }
        println("ZZ views $labels")
        val b0 = btn(labels[0]); val b1 = btn(labels[1]); val b2 = btn(labels[2])
        println("ZZ buttons ${b0.left}..${b0.right} ${b1.left}..${b1.right} ${b2.left}..${b2.right} y=${b1.center.y}")
        fun segLeft(): Int {
            val px = rule.onRoot().captureToImage().toPixelMap()
            val y = b1.center.y.toInt()
            for (x in (b0.left.toInt() - 30).coerceAtLeast(0) until (b2.right.toInt() + 30).coerceAtMost(px.width - 1)) {
                val c = px[x, y]
                if (kotlin.math.abs(c.red - volt.red) < .12f && kotlin.math.abs(c.green - volt.green) < .12f && kotlin.math.abs(c.blue - volt.blue) < .12f) return x
            }
            return -1
        }
        println("ZZ at rest: segment left=${segLeft()}")
        rule.mainClock.autoAdvance = false
        rule.onRoot().performTouchInput { click(b1.center) }
        androidx.compose.runtime.snapshots.Snapshot.sendApplyNotifications()
        val trail = mutableListOf<Int>()
        repeat(6) { rule.mainClock.advanceTimeBy(16); trail += segLeft() }
        println("ZZ pick 2, first 96ms: $trail")
        rule.onRoot().performTouchInput { click(b2.center) }
        androidx.compose.runtime.snapshots.Snapshot.sendApplyNotifications()
        val trail2 = mutableListOf<Int>()
        repeat(4) { rule.mainClock.advanceTimeBy(16); trail2 += segLeft() }
        println("ZZ pick 3 at 96ms, next 64ms: $trail2 (b1.left=${b1.left.toInt()})")
        rule.mainClock.autoAdvance = true
        org.junit.Assert.assertTrue("the segment jumped from ${trail.last()} to ${trail2.first()}", kotlin.math.abs(trail2.first() - trail.last()) < 40)
    }
}
